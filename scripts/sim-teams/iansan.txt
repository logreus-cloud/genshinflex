options swap_delay=12 iteration=1000;
target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

fn varesa_combo(amt) {
  for let j = 0; j < amt; j = j + 1 {
    if .varesa.skill.ready {
      varesa skill, attack, high_plunge[collision=1];
    } else if .xianyun.status.xianyun-airborne-buff {
      varesa jump;
      if .airborne {
        varesa high_plunge[collision=1];
      }
    } else {
      varesa charge, high_plunge[collision=1];
    }
    if .varesa.energy >= 30 && .varesa.status.apex-drive {
      varesa burst;
    }
  }
}

active furina;
for let i=0; i<4; i=i+1 {
  furina skill, dash, burst;
  iansan skill, charge, burst;
  xianyun attack, skill, low_plunge, burst;
  varesa_combo(7);
}
