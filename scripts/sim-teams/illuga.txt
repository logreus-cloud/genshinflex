columbina add set="silkenmoonsserenade" count=4;

yelan char lvl=90/90 cons=0 talent=9,9,9;

yelan add set="emblemofseveredfate" count=4;
yelan add stats hp=4780 atk=311 hp%=0.466 hydro%=0.466 cr=0.311;

options swap_delay=12 iteration=1000;
target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=999999999;
energy every interval=480,720 amount=1;

active navia;
for let i=0; i<5; i=i+1 {
  if .navia.burst.ready {
    navia burst;
  }
  columbina skill;
  if .columbina.burst.ready {
    columbina burst;
  }
  illuga skill;
  if .illuga.burst.ready {
    illuga burst;
  }
  yelan skill;
  if .yelan.burst.ready {
    yelan burst;
  }
  yelan attack;
  navia skill, attack:3, dash, attack:2;
}
